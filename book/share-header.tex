    leftline=true,
    rightline=false,
    topline=false,
    bottomline=false,
    linewidth=2pt,
    linecolor=sealred,
    innerleftmargin=14pt,
    innerrightmargin=4pt,
    innertopmargin=2pt,
    innerbottommargin=2pt,
    leftmargin=4pt,
    rightmargin=4pt,
    skipabove=8pt,
    skipbelow=8pt
  ]\color{quotegray}\itshape%
}{%
  \end{mdframed}%
}
